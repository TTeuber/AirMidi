# The engine on its own, without Flutter: a loopback through its own virtual port, with timing (README).
cmake_minimum_required(VERSION 3.20)
project(midi_engine_test CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(engine_test engine_test.cpp ../src/engine.cpp)
target_include_directories(engine_test PRIVATE ../src ../third_party/libremidi/include)
target_compile_definitions(engine_test PRIVATE LIBREMIDI_HEADER_ONLY)
if(APPLE)
  target_compile_definitions(engine_test PRIVATE LIBREMIDI_COREMIDI)
  target_link_libraries(engine_test PRIVATE "-framework CoreMIDI" "-framework CoreAudio" "-framework CoreFoundation")
elseif(WIN32)
  target_compile_definitions(engine_test PRIVATE LIBREMIDI_WINMM)
  target_link_libraries(engine_test PRIVATE winmm)
else()
  target_compile_definitions(engine_test PRIVATE LIBREMIDI_ALSA)
  find_package(ALSA REQUIRED)
  target_link_libraries(engine_test PRIVATE ALSA::ALSA pthread)
endif()
